## 1. 线段树 (Segment Tree)

### 原理
线段树是一种二叉树结构，每个节点存储一个区间的信息。对于区间最值问题：
- 叶子节点存储单个元素的值
- 内部节点存储子节点值的最大值/最小值
- 支持区间查询和单点/区间更新

### Python实现
```python
class SegmentTree:
    def __init__(self, data):
        self.n = len(data)
        self.size = 1
        while self.size < self.n:
            self.size *= 2
        # 初始化线段树数组
        self.max_tree = [0] * (2 * self.size)
        self.min_tree = [0] * (2 * self.size)
        
        # 构建叶子节点
        for i in range(self.n):
            self.max_tree[self.size + i] = data[i]
            self.min_tree[self.size + i] = data[i]
        
        # 构建内部节点
        for i in range(self.size - 1, 0, -1):
            self.max_tree[i] = max(self.max_tree[2 * i], self.max_tree[2 * i + 1])
            self.min_tree[i] = min(self.min_tree[2 * i], self.min_tree[2 * i + 1])
    
    def update(self, index, value):
        """更新指定位置的值"""
        pos = self.size + index
        self.max_tree[pos] = value
        self.min_tree[pos] = value
        
        # 向上更新父节点
        pos //= 2
        while pos > 0:
            self.max_tree[pos] = max(self.max_tree[2 * pos], self.max_tree[2 * pos + 1])
            self.min_tree[pos] = min(self.min_tree[2 * pos], self.min_tree[2 * pos + 1])
            pos //= 2
    
    def query_range_max(self, l, r):
        """查询区间[l, r]的最大值"""
        l += self.size
        r += self.size
        max_val = float('-inf')
        
        while l <= r:
            if l % 2 == 1:  # l是右子节点
                max_val = max(max_val, self.max_tree[l])
                l += 1
            if r % 2 == 0:  # r是左子节点
                max_val = max(max_val, self.max_tree[r])
                r -= 1
            l //= 2
            r //= 2
        
        return max_val
    
    def query_range_min(self, l, r):
        """查询区间[l, r]的最小值"""
        l += self.size
        r += self.size
        min_val = float('inf')
        
        while l <= r:
            if l % 2 == 1:  # l是右子节点
                min_val = min(min_val, self.min_tree[l])
                l += 1
            if r % 2 == 0:  # r是左子节点
                min_val = min(min_val, self.min_tree[r])
                r -= 1
            l //= 2
            r //= 2
        
        return min_val

# 使用示例
data = [1, 3, 2, 7, 9, 11, 5, 8]
seg_tree = SegmentTree(data)

print(f"数组: {data}")
print(f"区间[2,5]的最大值: {seg_tree.query_range_max(2, 5)}")  # 输出: 11
print(f"区间[2,5]的最小值: {seg_tree.query_range_min(2, 5)}")  # 输出: 2

seg_tree.update(3, 1)  # 将索引3的值从7改为1
print(f"更新后区间[2,5]的最大值: {seg_tree.query_range_max(2, 5)}")  # 输出: 11
print(f"更新后区间[2,5]的最小值: {seg_tree.query_range_min(2, 5)}")  # 输出: 1
```

## 2. 树状数组 (Fenwick Tree / Binary Indexed Tree)

### 原理
树状数组利用二进制索引来高效维护前缀信息。对于区间最值：
- 每个节点存储一定区间的最值
- 通过lowbit操作快速定位相关区间
- 更新和查询的时间复杂度都是O(log n)

### Python实现
```python
class FenwickTree:
    def __init__(self, data):
        self.n = len(data)
        self.arr = data[:]
        self.max_tree = [0] * (self.n + 1)
        self.min_tree = [float('inf')] * (self.n + 1)
        
        # 初始化树状数组
        for i in range(self.n):
            self._update(i, self.arr[i])
    
    def _lowbit(self, x):
        return x & -x
    
    def _update(self, index, value):
        """更新树状数组"""
        pos = index + 1
        
        # 更新最大值树
        while pos <= self.n:
            if value > self.max_tree[pos]:
                self.max_tree[pos] = value
            else:
                # 需要重新计算该位置的最大值
                current_max = self.arr[index]
                r = pos - self._lowbit(pos) + 1
                for i in range(r, pos):
                    if self.arr[i-1] > current_max:
                        current_max = self.arr[i-1]
                self.max_tree[pos] = current_max
            pos += self._lowbit(pos)
        
        # 更新最小值树
        pos = index + 1
        while pos <= self.n:
            if value < self.min_tree[pos]:
                self.min_tree[pos] = value
            else:
                # 需要重新计算该位置的最小值
                current_min = self.arr[index]
                r = pos - self._lowbit(pos) + 1
                for i in range(r, pos):
                    if self.arr[i-1] < current_min:
                        current_min = self.arr[i-1]
                self.min_tree[pos] = current_min
            pos += self._lowbit(pos)
    
    def update(self, index, value):
        """更新指定位置的值"""
        self.arr[index] = value
        self._update(index, value)
    
    def query_prefix_max(self, index):
        """查询前缀[0, index]的最大值"""
        pos = index + 1
        max_val = float('-inf')
        
        while pos > 0:
            max_val = max(max_val, self.max_tree[pos])
            pos -= self._lowbit(pos)
        
        return max_val
    
    def query_prefix_min(self, index):
        """查询前缀[0, index]的最小值"""
        pos = index + 1
        min_val = float('inf')
        
        while pos > 0:
            min_val = min(min_val, self.min_tree[pos])
            pos -= self._lowbit(pos)
        
        return min_val
    
    def query_range_max(self, l, r):
        """查询区间[l, r]的最大值"""
        if l == 0:
            return self.query_prefix_max(r)
        
        max_val = self.arr[r]
        current = r
        
        while current >= l:
            if current - self._lowbit(current) + 1 >= l:
                max_val = max(max_val, self.max_tree[current + 1])
                current -= self._lowbit(current)
            else:
                max_val = max(max_val, self.arr[current])
                current -= 1
        
        return max_val

# 使用示例
data = [1, 3, 2, 7, 9, 11, 5, 8]
fenwick_tree = FenwickTree(data)

print(f"数组: {data}")
print(f"前缀[0,5]的最大值: {fenwick_tree.query_prefix_max(5)}")  # 输出: 11
print(f"前缀[0,5]的最小值: {fenwick_tree.query_prefix_min(5)}")  # 输出: 1
print(f"区间[2,5]的最大值: {fenwick_tree.query_range_max(2, 5)}")  # 输出: 11
```

## 3. 二叉树实现

### 原理
使用平衡二叉树（如AVL树）来维护序列，每个节点存储子树的最值信息。

### Python实现
```python
class TreeNode:
    def __init__(self, val, index):
        self.val = val
        self.index = index
        self.left = None
        self.right = None
        self.height = 1
        self.subtree_max = val
        self.subtree_min = val

class BSTWithRangeQuery:
    def __init__(self):
        self.root = None
        self.nodes = []  # 存储所有节点引用
    
    def _get_height(self, node):
        if not node:
            return 0
        return node.height
    
    def _get_balance(self, node):
        if not node:
            return 0
        return self._get_height(node.left) - self._get_height(node.right)
    
    def _update_node_info(self, node):
        """更新节点的高度和子树最值"""
        if not node:
            return
        
        node.height = 1 + max(self._get_height(node.left), 
                            self._get_height(node.right))
        
        node.subtree_max = node.val
        node.subtree_min = node.val
        
        if node.left:
            node.subtree_max = max(node.subtree_max, node.left.subtree_max)
            node.subtree_min = min(node.subtree_min, node.left.subtree_min)
        if node.right:
            node.subtree_max = max(node.subtree_max, node.right.subtree_max)
            node.subtree_min = min(node.subtree_min, node.right.subtree_min)
    
    def _rotate_right(self, y):
        """右旋转"""
        x = y.left
        T2 = x.right
        
        x.right = y
        y.left = T2
        
        self._update_node_info(y)
        self._update_node_info(x)
        
        return x
    
    def _rotate_left(self, x):
        """左旋转"""
        y = x.right
        T2 = y.left
        
        y.left = x
        x.right = T2
        
        self._update_node_info(x)
        self._update_node_info(y)
        
        return y
    
    def insert(self, val, index):
        """插入节点"""
        self.root = self._insert(self.root, val, index)
    
    def _insert(self, node, val, index):
        if not node:
            new_node = TreeNode(val, index)
            self.nodes.append(new_node)
            return new_node
        
        if index < node.index:
            node.left = self._insert(node.left, val, index)
        else:
            node.right = self._insert(node.right, val, index)
        
        self._update_node_info(node)
        
        balance = self._get_balance(node)
        
        # 左左情况
        if balance > 1 and index < node.left.index:
            return self._rotate_right(node)
        
        # 右右情况
        if balance < -1 and index > node.right.index:
            return self._rotate_left(node)
        
        # 左右情况
        if balance > 1 and index > node.left.index:
            node.left = self._rotate_left(node.left)
            return self._rotate_right(node)
        
        # 右左情况
        if balance < -1 and index < node.right.index:
            node.right = self._rotate_right(node.right)
            return self._rotate_left(node)
        
        return node
    
    def _query_range_util(self, node, l, r):
        """查询区间[l, r]的最值"""
        if not node or node.index < l or node.index > r:
            return float('-inf'), float('inf')
        
        # 如果当前节点的区间完全包含在查询区间内
        if (not node.left or node.left.index >= l) and \
           (not node.right or node.right.index <= r):
            return node.subtree_max, node.subtree_min
        
        left_max, left_min = self._query_range_util(node.left, l, r)
        right_max, right_min = self._query_range_util(node.right, l, r)
        
        current_val = node.val if l <= node.index <= r else float('-inf'), float('inf')
        
        max_val = max(left_max, right_max)
        min_val = min(left_min, right_min)
        
        if l <= node.index <= r:
            max_val = max(max_val, node.val)
            min_val = min(min_val, node.val)
        
        return max_val, min_val
    
    def query_range(self, l, r):
        """查询区间[l, r]的最大值和最小值"""
        return self._query_range_util(self.root, l, r)

# 使用示例
bst = BSTWithRangeQuery()
data = [1, 3, 2, 7, 9, 11, 5, 8]

# 构建二叉树
for i, val in enumerate(data):
    bst.insert(val, i)

max_val, min_val = bst.query_range(2, 5)
print(f"数组: {data}")
print(f"区间[2,5]的最大值: {max_val}, 最小值: {min_val}")  # 输出: 最大值: 11, 最小值: 2
```

## 性能比较

| 数据结构 | 构建时间 | 查询时间 | 更新时间 | 空间复杂度 | 适用场景 |
|---------|----------|----------|----------|------------|----------|
| 线段树 | O(n) | O(log n) | O(log n) | O(n) | 通用，支持各种区间操作 |
| 树状数组 | O(n log n) | O(log n) | O(log n) | O(n) | 前缀查询，代码简洁 |
| 二叉树 | O(n log n) | O(log n) | O(log n) | O(n) | 动态插入删除 |

## 总结

1. **线段树**：最通用，支持各种区间操作，代码稍复杂但功能强大
2. **树状数组**：代码简洁，适合前缀查询，但区间最值实现相对复杂
3. **二叉树**：适合动态数据集，支持插入删除操作

根据具体需求选择合适的结构：
- 静态数据+频繁查询：线段树
- 前缀查询为主：树状数组  
- 动态数据+查询：平衡二叉树
# 支持动态操作的树结构实现

当需要支持修改、增加、删除节点时，我们需要对之前的树结构进行改进。下面我提供支持完整动态操作的实现：

## 1. 动态线段树

```python
class DynamicSegmentTree:
    def __init__(self, data=None):
        self.data = data if data else []
        if data:
            self._build_tree()
        else:
            self.root = None
    
    class Node:
        def __init__(self, start, end, val=None):
            self.start = start
            self.end = end
            self.left = None
            self.right = None
            self.max_val = val
            self.min_val = val
            self.val = val  # 对于叶子节点存储实际值
    
    def _build_tree(self, start=0, end=None):
        """递归构建线段树"""
        if end is None:
            end = len(self.data) - 1
        
        if start == end:
            return self.Node(start, end, self.data[start])
        
        mid = (start + end) // 2
        node = self.Node(start, end)
        node.left = self._build_tree(start, mid)
        node.right = self._build_tree(mid + 1, end)
        
        node.max_val = max(node.left.max_val, node.right.max_val)
        node.min_val = min(node.left.min_val, node.right.min_val)
        
        return node
    
    def _update_node(self, node):
        """更新节点的最值信息"""
        if node.left and node.right:
            node.max_val = max(node.left.max_val, node.right.max_val)
            node.min_val = min(node.left.min_val, node.right.min_val)
        elif node.left:
            node.max_val = node.left.max_val
            node.min_val = node.left.min_val
        elif node.right:
            node.max_val = node.right.max_val
            node.min_val = node.right.min_val
    
    def update(self, index, value, node=None):
        """更新指定位置的值"""
        if node is None:
            node = self.root
        
        if node.start == node.end == index:
            node.val = value
            node.max_val = value
            node.min_val = value
            return
        
        mid = (node.start + node.end) // 2
        
        if index <= mid:
            self.update(index, value, node.left)
        else:
            self.update(index, value, node.right)
        
        self._update_node(node)
    
    def insert(self, index, value):
        """在指定位置插入值"""
        if not self.data:
            self.data = [value]
            self.root = self.Node(0, 0, value)
            return
        
        # 更新数据数组
        self.data.insert(index, value)
        
        # 重新构建树（简单实现，实际可以增量更新）
        self._rebuild_tree()
    
    def delete(self, index):
        """删除指定位置的值"""
        if not self.data or index >= len(self.data):
            return
        
        # 更新数据数组
        self.data.pop(index)
        
        # 重新构建树
        if self.data:
            self._rebuild_tree()
        else:
            self.root = None
    
    def _rebuild_tree(self):
        """重新构建整棵树"""
        self.root = self._build_tree()
    
    def query_range(self, l, r, node=None):
        """查询区间[l, r]的最值"""
        if node is None:
            node = self.root
        
        # 当前节点区间完全在查询区间内
        if l <= node.start and node.end <= r:
            return node.max_val, node.min_val
        
        mid = (node.start + node.end) // 2
        max_val = float('-inf')
        min_val = float('inf')
        
        # 查询左子树
        if l <= mid and node.left:
            left_max, left_min = self.query_range(l, r, node.left)
            max_val = max(max_val, left_max)
            min_val = min(min_val, left_min)
        
        # 查询右子树
        if r > mid and node.right:
            right_max, right_min = self.query_range(l, r, node.right)
            max_val = max(max_val, right_max)
            min_val = min(min_val, right_min)
        
        return max_val, min_val

# 使用示例
print("=== 动态线段树 ===")
data = [1, 3, 2, 7, 9]
ds_tree = DynamicSegmentTree(data)
print(f"初始数组: {ds_tree.data}")
print(f"区间[1,3]的最值: {ds_tree.query_range(1, 3)}")

print(f"插入后数组: {ds_tree.data}")
print(f"插入后区间[1,3]的最值: {ds_tree.query_range(1, 3)}")

print(f"更新后数组: {ds_tree.data}")
print(f"更新后区间[1,3]的最值: {ds_tree.query_range(1, 3)}")

ds_tree.delete(2)  # 删除索引2的值
print(f"删除后数组: {ds_tree.data}")
```

## 2. 支持动态操作的平衡二叉树

```python
class DynamicBST:
    def __init__(self):
        self.root = None
        self.size = 0
    
    class TreeNode:
        def __init__(self, val, index):
            self.val = val
            self.index = index  # 在序列中的位置
            self.left = None
            self.right = None
            self.height = 1
            self.size = 1  # 子树大小
            self.max_val = val
            self.min_val = val
    
    def _get_height(self, node):
        return node.height if node else 0
    
    def _get_size(self, node):
        return node.size if node else 0
    
    def _get_balance(self, node):
        if not node:
            return 0
        return self._get_height(node.left) - self._get_height(node.right)
    
    def _update_node(self, node):
        """更新节点的高度、大小和最值信息"""
        if not node:
            return
        
        node.height = 1 + max(self._get_height(node.left), 
                            self._get_height(node.right))
        node.size = 1 + self._get_size(node.left) + self._get_size(node.right)
        
        node.max_val = node.val
        node.min_val = node.val
        
        if node.left:
            node.max_val = max(node.max_val, node.left.max_val)
            node.min_val = min(node.min_val, node.left.min_val)
        if node.right:
            node.max_val = max(node.max_val, node.right.max_val)
            node.min_val = min(node.min_val, node.right.min_val)
    
    def _rotate_right(self, y):
        """右旋转"""
        x = y.left
        T2 = x.right
        
        x.right = y
        y.left = T2
        
        self._update_node(y)
        self._update_node(x)
        
        return x
    
    def _rotate_left(self, x):
        """左旋转"""
        y = x.right
        T2 = y.left
        
        y.left = x
        x.right = T2
        
        self._update_node(x)
        self._update_node(y)
        
        return y
    
    def _balance_node(self, node):
        """平衡节点"""
        balance = self._get_balance(node)
        
        # 左左情况
        if balance > 1 and self._get_balance(node.left) >= 0:
            return self._rotate_right(node)
        
        # 右右情况
        if balance < -1 and self._get_balance(node.right) <= 0:
            return self._rotate_left(node)
        
        # 左右情况
        if balance > 1 and self._get_balance(node.left) < 0:
            node.left = self._rotate_left(node.left)
            return self._rotate_right(node)
        
        # 右左情况
        if balance < -1 and self._get_balance(node.right) > 0:
            node.right = self._rotate_right(node.right)
            return self._rotate_left(node)
        
        return node
    
    def insert_at_index(self, index, value):
        """在指定位置插入值"""
        self.root = self._insert_at_index(self.root, index, value)
        self.size += 1
    
    def _insert_at_index(self, node, index, value):
        if not node:
            return self.TreeNode(value, index)
        
        left_size = self._get_size(node.left)
        
        if index <= left_size:
            node.left = self._insert_at_index(node.left, index, value)
        else:
            node.right = self._insert_at_index(node.right, index - left_size - 1, value)
        
        self._update_node(node)
        return self._balance_node(node)
    
    def delete_at_index(self, index):
        """删除指定位置的值"""
        if index < 0 or index >= self.size:
            return None
        
        self.root, deleted = self._delete_at_index(self.root, index)
        if deleted:
            self.size -= 1
        return deleted
    
    def _delete_at_index(self, node, index):
        if not node:
            return node, None
        
        left_size = self._get_size(node.left)
        
        if index < left_size:
            node.left, deleted = self._delete_at_index(node.left, index)
        elif index > left_size:
            node.right, deleted = self._delete_at_index(node.right, index - left_size - 1)
        else:
            # 找到要删除的节点
            if not node.left:
                return node.right, node
            elif not node.right:
                return node.left, node
            
            # 有两个子节点，找到中序后继
            temp = self._min_value_node(node.right)
            node.val = temp.val
            node.index = temp.index
            node.right, _ = self._delete_at_index(node.right, 0)
            deleted = node
        
        self._update_node(node)
        return self._balance_node(node), deleted
    
    def _min_value_node(self, node):
        """找到子树中的最小节点"""
        current = node
        while current.left:
            current = current.left
        return current
    
    def update_at_index(self, index, value):
        """更新指定位置的值"""
        node = self._find_node_by_index(self.root, index)
        if node:
            node.val = value
            # 更新后需要重新计算最值
            self._update_tree_from_node(node)
            return True
        return False
    
    def _find_node_by_index(self, node, index):
        """根据索引查找节点"""
        if not node:
            return None
        
        left_size = self._get_size(node.left)
        
        if index < left_size:
            return self._find_node_by_index(node.left, index)
        elif index > left_size:
            return self._find_node_by_index(node.right, index - left_size - 1)
        else:
            return node
    
    def _update_tree_from_node(self, node):
        """从指定节点开始向上更新树"""
        while node:
            self._update_node(node)
            # 在实际实现中，这里需要找到父节点
            # 为简化，我们重新计算整棵树
            node = None
        
        # 重新计算整棵树的最值
        self._update_all_nodes(self.root)
    
    def _update_all_nodes(self, node):
        """递归更新所有节点"""
        if not node:
            return
        self._update_all_nodes(node.left)
        self._update_all_nodes(node.right)
        self._update_node(node)
    
    def query_range(self, l, r):
        """查询区间[l, r]的最值"""
        if l > r or l < 0 or r >= self.size:
            return float('-inf'), float('inf')
        
        return self._query_range(self.root, l, r)
    
    def _query_range(self, node, l, r):
        if not node or node.index < l or node.index > r:
            return float('-inf'), float('inf')
        
        # 如果当前节点的区间完全包含在查询区间内
        left_size = self._get_size(node.left)
        node_index = left_size
        
        # 检查当前节点是否在区间内
        current_max = node.val if l <= node_index <= r else float('-inf')
        current_min = node.val if l <= node_index <= r else float('inf')
        
        # 查询左子树和右子树
        left_max, left_min = self._query_range(node.left, l, r)
        right_max, right_min = self._query_range(node.right, l, r)
        
        max_val = max(left_max, right_max, current_max)
        min_val = min(left_min, right_min, current_min)
        
        return max_val, min_val
    
    def to_list(self):
        """将树转换为列表"""
        result = []
        self._inorder_traversal(self.root, result)
        return result
    
    def _inorder_traversal(self, node, result):
        if not node:
            return
        self._inorder_traversal(node.left, result)
        result.append(node.val)
        self._inorder_traversal(node.right, result)
```

## 3. 更高效的动态线段树（延迟构建）

```python
class LazyDynamicSegmentTree:
    def __init__(self):
        self.root = None
    
    class Node:
        def __init__(self, start, end):
            self.start = start
            self.end = end
            self.left = None
            self.right = None
            self.max_val = float('-inf')
            self.min_val = float('inf')
            self.val = None
            self.lazy = None  # 延迟标记
    
    def _ensure_node(self, start, end):
        """确保节点存在，延迟构建"""
        if not self.root:
            self.root = self.Node(start, end)
        return self.root
    
    def _split_node(self, node, mid):
        """分割节点"""
        if not node.left and node.start != mid:
            node.left = self.Node(node.start, mid)
        if not node.right and node.end != mid + 1:
            node.right = self.Node(mid + 1, node.end)
    
    def update(self, index, value, node=None, start=None, end=None):
        """更新指定位置的值"""
        if node is None:
            node = self._ensure_node(index, index)
            start, end = node.start, node.end
        
        if start == end == index:
            node.val = value
            node.max_val = value
            node.min_val = value
            return
        
        mid = (start + end) // 2
        self._split_node(node, mid)
        
        if index <= mid:
            self.update(index, value, node.left, start, mid)
        else:
            self.update(index, value, node.right, mid + 1, end)
        
        # 更新父节点的最值
        node.max_val = max(
            node.left.max_val if node.left else float('-inf'),
            node.right.max_val if node.right else float('-inf')
        )
        node.min_val = min(
            node.left.min_val if node.left else float('inf'),
            node.right.min_val if node.right else float('inf')
        )
    
    def query_range(self, l, r, node=None, start=None, end=None):
        """查询区间[l, r]的最值"""
        if node is None:
            node = self.root
            if not node:
                return float('-inf'), float('inf')
            start, end = node.start, node.end
        
        # 当前节点区间完全在查询区间外
        if end < l or start > r:
            return float('-inf'), float('inf')
        
        # 当前节点区间完全在查询区间内
        if l <= start and end <= r:
            return node.max_val, node.min_val
        
        mid = (start + end) // 2
        max_val = float('-inf')
        min_val = float('inf')
        
        if node.left and l <= mid:
            left_max, left_min = self.query_range(l, r, node.left, start, mid)
            max_val = max(max_val, left_max)
            min_val = min(min_val, left_min)
        
        if node.right and r > mid:
            right_max, right_min = self.query_range(l, r, node.right, mid + 1, end)
            max_val = max(max_val, right_max)
            min_val = min(min_val, right_min)
        
        return max_val, min_val

# 使用示例
print("\n=== 延迟构建动态线段树 ===")
ldst = LazyDynamicSegmentTree()

# 动态更新和查询
ldst.update(0, 1)
ldst.update(1, 3)
ldst.update(2, 2)
ldst.update(3, 7)
ldst.update(4, 9)

print(f"区间[1,3]的最值: {ldst.query_range(1, 3)}")

ldst.update(2, 5)  # 更新索引2的值为5
print(f"更新后区间[1,3]的最值: {ldst.query_range(1, 3)}")

ldst.update(5, 11)  # 在索引5插入11（动态扩展）
print(f"扩展后区间[3,5]的最值: {ldst.query_range(3, 5)}")
```

## 性能比较和选择建议

| 操作 | 动态线段树 | 平衡二叉树 | 延迟构建线段树 |
| ---- | ---------- | ---------- | -------------- |
| 查询 | O(log n)   | O(log n)   | O(log n)       |
| 更新 | O(log n)   | O(log n)   | O(log n)       |
| 插入 | O(n) 重建  | O(log n)   | O(log n)       |
| 删除 | O(n) 重建  | O(log n)   | 不支持直接删除 |
| 空间 | O(n)       | O(n)       | O(动态)        |

**选择建议：**
- **频繁插入删除**：使用平衡二叉树
- **主要查询和更新，偶尔插入**：使用动态线段树
- **稀疏数据或范围很大**：使用延迟构建线段树
- **固定大小数据**：使用普通线段树或树状数组

这些实现都支持动态修改，你可以根据具体的使用场景选择合适的实现。
